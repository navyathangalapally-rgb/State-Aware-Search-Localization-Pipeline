import os
import time
import shutil
from bs4 import BeautifulSoup
from deep_translator import GoogleTranslator

# --- SETTINGS ---
SOURCE_DIR = 'DE.HTML' 
OUTPUT_DIR = 'GERMAN_DONE'
TARGET_LANG = 'de'
# ----------------

def translate_file(filename):
    translator = GoogleTranslator(source='en', target=TARGET_LANG)
    input_path = os.path.join(SOURCE_DIR, filename)
    
    with open(input_path, 'r', encoding='utf-8') as f:
        soup = BeautifulSoup(f, 'html.parser')

    # FIX: Ensure tables and the searchable wrapper allow full width
    if soup.head:
        extra_style = soup.new_tag('style')
        extra_style.string = "#help-content-wrapper { width: 100%; } table { display: table !important; }"
        soup.head.append(extra_style)

    # FIX: Update HTML links to find your local assets
    for link in soup.find_all('link', rel='stylesheet'):
        link['href'] = 'styles.css'
    for script in soup.find_all('script', src=True):
        if 'searchbar' in script['src']:
            script['src'] = os.path.basename(script['src'])

    # TRANSLATION
    nodes = []
    texts = []
    for element in soup.find_all(string=True):
        if element.parent.name not in ['style', 'script', 'head', 'title', 'meta']:
            clean = element.strip()
            if clean and len(clean) > 1 and any(c.isalpha() for c in clean):
                nodes.append(element)
                texts.append(clean)

    if texts:
        try:
            # Slower batching (groups of 5) to avoid Google blocking you
            for i in range(0, len(texts), 5):
                chunk = texts[i:i+5]
                translated = translator.translate_batch(chunk)
                for j, t_text in enumerate(translated):
                    nodes[i+j].replace_with(t_text)
                time.sleep(1.5) # Pause to stay safe
        except:
            pass 

    with open(os.path.join(OUTPUT_DIR, filename), 'w', encoding='utf-8') as f:
        f.write(str(soup))

if __name__ == "__main__":
    if not os.path.exists(OUTPUT_DIR): os.makedirs(OUTPUT_DIR)
    
    # Copy Assets
    for asset in ['styles.css', 'searchbar1.js', 'searchbar2.js']:
        if os.path.exists(asset): shutil.copy2(asset, os.path.join(OUTPUT_DIR, asset))

    files = [f for f in os.listdir(SOURCE_DIR) if f.lower().endswith('.html')]
    print(f"Starting final fix for {len(files)} files...")
    for i, f in enumerate(files, 1):
        print(f"[{i}/{len(files)}] Processing: {f}")
        translate_file(f)
        time.sleep(2)
    print("✅ Done!")
